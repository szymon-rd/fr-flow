/** Writes the rooms module as the printer gives it, for `lake` (r43-spike/rooms-lean-r15): not a suite of the build. */
class LeanDump extends munit.FunSuite:
  test("dump") {
    val examples = _root_.rooms.roomsExamples.elaborated.fold(e => fail(e.toString), identity)
    val w = Whole(_root_.rooms.rooms, examples)
    val r = w.resolved
    val table = TermTable.of(r)
    val v = r.theory.vocabulary
    def opens(d: DefName) = table.defines.get(d).toList.flatMap(_.symbols.map(s => Names.ident(s.value)))
    val env = LowerEnv(KbName("Rooms"), opens, _ => Names.builtin("store"), _ => Names.builtin("Bookings"),
      n => v.dataType(n))
    val naming = Naming.over(Set("granted", "book"), Set("Booking", "RoomBookings", "World"))
    val defs = List("endsAfterStart", "covers", "apart", "isFree", "noOverlap", "neverOverlap", "roomBookingsApart",
        "startsEmpty").map { n =>
      val d = v.define(DefName(n)).get
      d.written.form match
        case Form.Term(t) =>
          val (bs, body) = Printed.define(env, naming)(d.lambda.params, t.run)
          s"@[frflow_defs] def $n ${Print.binders(bs)} : ${Print.expr(LowerType(d.lambda.result))} := ${Print.expr(body)}"
        case _ => ""
    }
    val worlds = v.models.flatMap(m => v.boundary(m.boundary).map(_ -> m))
    val models = LowerModels(env, naming)
    val prog = v.define(DefName("bookIfGranted")).get
    val progBody = prog.written.form match
      case Form.Steps(p) =>
        val inputs = prog.lambda.params
        val (bs, lines) = (inputs.traverse(x => Naming.bind(x).map(LBinder.Explicit(_, LowerType(x.ty)))),
          LowerProg(env, inputs.size).block(p)).tupled.runA(naming).value
        Print.expr(LExpr.Fun(bs, LExpr.Do(lines, 4)))
      case _ => ""
    val run = v.define(DefName("bookIfGranted_run")).map(d => models.run(d, DefName("bookIfGranted"),
      table.defines(d.name).symbols.map(s => s -> table.typeOf(s)))).get
    val guards = Guards.of(r.theory)
    def ps(id: NodeId) = table.claims(id).symbols.map(s =>
      s"(${s.value} : ${Print.expr(LowerType.binder(table.typeOf(s)))})").mkString(" ")
    val claimed = List("grantWhenNoOverlap", "grantWhenFree", "bookWhenFree", "answersWhetherBooked").map { n =>
      val f = r.theory.node(NodeId(n)).flatMap(_.statement.said).get
      s"namespace $n\n@[frflow_defs] def claim ${ps(NodeId(n))} : Prop := " +
        s"${Print.expr(Printed.claim(env, naming, guards)(f.term))}\nend $n"
    }
    val keeps = List("neverOverlap", "endsAfterStart").flatMap(n => Required.keeps(r, NodeId(n)).map(n -> _))
    val keepDecls = keeps.map(_._2.goal).collect { case Prop.Forall(_, Prop.Implies(_, k: Prop.Keeps)) =>
      PrintDecl(LowerKeeps(env, table, naming)(k)) }
    val made = List("roomBookingsApart", "endsAfterStart").flatMap(n => Required.made(r, NodeId(n)).map(n -> _))
    val madeDecls = made.map(_._2.goal).collect { case Prop.Forall(_, k: Prop.Makes) =>
      PrintDecl(LowerKeeps(env, table, naming).makes(k)) }
    val results = v.models.flatMap(m => v.boundary(m.boundary).toList.flatMap(_.ops.filter(_.result != frflow.core.ty.Ty.Unit))
      .map(o => PrintDecl(models.resultOf(m, o))))
    def inNode(n: String, os: List[Obligation]) =
      s"namespace $n\n\n" + os.map(o => PrintDecl(LowerObligation(LowerProp(table, env))(o))).mkString("\n\n") +
        s"\n\nend $n"
    val obls = List(inNode("bookWhenFree", Required.impl(r, NodeId("bookWhenFree"))),
      inNode("answersWhetherBooked", Required.impl(r, NodeId("answersWhetherBooked")))) ++
      keeps.map((n, o) => inNode(n, List(o))) ++ made.map((n, o) => inNode(n, List(o)))
    val whole = WholeModule(LowerProp(table, env), LowerWitness(env, Naming.over(Set.empty, Set.empty)))
      .of(KbName("Rooms"), Nil, w.obligations).decls.map(PrintDecl(_))
    val out = List("import FrFlow", "set_option autoImplicit false", "set_option linter.unusedVariables false",
      "noncomputable section", "open Classical", "namespace Rooms",
      "structure Booking where\n  room : String\n  start : Int\n  «end» : Int\nderiving DecidableEq",
      "structure RoomBookings where\n  bookings : (List Booking)\nderiving DecidableEq") ++ defs ++
      List(models.decls(worlds).map(PrintDecl(_)).mkString("\n\n")) ++ results ++ List(
        s"@[frflow_defs] def bookIfGranted {m : Type → Type} [Monad m] [Bookings m] " +
          s"(granted : RoomBookings → Booking → Prop) : Booking → m Prop :=\n  $progBody",
        PrintDecl(run)) ++ claimed ++ List("end Rooms") ++ keepDecls ++ madeDecls ++ List("namespace Rooms") ++ obls ++
        List("namespace Whole") ++ whole ++ List("end Whole", "end Rooms")
    java.nio.file.Files.writeString(java.nio.file.Path.of(
      "/tmp/claude-0/-home-user-fr-flow/80a77cee-12ee-5106-9f28-3411c00b75ee/scratchpad/r43-spike/rooms-lean-r15/Dump.lean"),
      out.mkString("\n\n") + "\n")
  }

/** Writes auth's claims and `Whole.lean` as the printer gives them, for `lake` (r43-spike/rooms-lean-r15/AuthDump). */
class AuthDump extends munit.FunSuite:
  test("dump") {
    val w = Whole(auth.knowledge, Nil)
    val r = w.resolved
    val table = TermTable.of(r)
    val v = r.theory.vocabulary
    def opens(d: DefName) = table.defines.get(d).toList.flatMap(_.symbols.map(s => Names.ident(s.value)))
    val env = LowerEnv(KbName("Auth"), opens, _ => Names.builtin("store"), _ => Names.builtin("Store"),
      n => v.dataType(n))
    val naming = Naming.over(v.symbols.map(_.name.value).toSet, Set("UserId", "Path", "Grant", "Session"))
    val admits = v.define(DefName("admits")).get
    val admitsDef = admits.written.form match
      case Form.Term(t) =>
        val (bs, body) = Printed.define(env, naming)(admits.lambda.params, t.run)
        s"@[frflow_defs] def admits ${opens(admits.name).map(o => s"(${o.text} : ${Print.expr(LowerType.binder(
          table.typeOf(frflow.core.name.SymbolName(o.text))))})").mkString(" ")} ${Print.binders(bs)} : Prop := " +
          Print.expr(body)
      case _ => ""
    def ps(id: NodeId) = table.claims(id).symbols.map(s =>
      s"(${s.value} : ${Print.expr(LowerType.binder(table.typeOf(s)))})").mkString(" ")
    val claimed = List("accessExactly", "grantsAuthorize", "sessionsSignIn", "requestsReach").map { n =>
      val f = r.theory.node(NodeId(n)).flatMap(_.statement.said).get
      s"namespace $n\n@[frflow_defs] def claim ${ps(NodeId(n))} : Prop := " +
        s"${Print.expr(Printed.claim(env, naming, Guards.of(r.theory))(f.term))}\nend $n"
    }
    val whole = WholeModule(LowerProp(table, env), LowerWitness(env, Naming.over(Set.empty, Set.empty)))
      .of(KbName("Auth"), Nil, w.obligations).decls.map(PrintDecl(_))
    // R14.6: the three bottom claims of one group, solved in substitution order.
    val signedInBySessions = Node.claim("A user is signed in exactly when a session is open for them.")(
      forall((u: Term[UserId]) => signedIn(u) === sessions.exists(s => s.user === u)))
    val authorizedByGrants = Node.claim("A user is authorized for a path exactly when a grant names them and it.")(
      forall((u: Term[UserId], p: Term[Path]) => authorized(u, p) === grants.exists(g => g === make[Grant](u, p))))
    val b = Whole(KnowledgeBase("Auth")(accessExactly, signedInBySessions, authorizedByGrants)
      .implementProvided(grants, sessions), Nil)
    val bt = TermTable.of(b.resolved)
    def bps(id: NodeId) = bt.claims(id).symbols.map(s =>
      s"(${s.value} : ${Print.expr(LowerType.binder(bt.typeOf(s)))})").mkString(" ")
    val bottomClaims = List("signedInBySessions", "authorizedByGrants").map { n =>
      val f = b.resolved.theory.node(NodeId(n)).flatMap(_.statement.said).get
      s"namespace $n\n@[frflow_defs] def claim ${bps(NodeId(n))} : Prop := " +
        s"${Print.expr(Printed.claim(env, naming, Guards.of(b.resolved.theory))(f.term))}\nend $n"
    }
    val bottom = WholeModule(LowerProp(bt, env), LowerWitness(env, Naming.over(Set.empty, Set.empty)))
      .of(KbName("Auth"), Nil, b.obligations).decls.map(PrintDecl(_))
    val out = List("import FrFlow", "set_option autoImplicit false", "set_option linter.unusedVariables false",
      "noncomputable section", "open Classical", "namespace Auth",
      "structure UserId where\n  name : String\nderiving DecidableEq",
      "structure Path where\n  value : String\nderiving DecidableEq",
      "structure Grant where\n  user : UserId\n  path : Path\nderiving DecidableEq",
      "structure Session where\n  token : String\n  user : UserId\nderiving DecidableEq", admitsDef) ++ claimed ++
      List("namespace Whole") ++ whole ++ List("end Whole") ++ bottomClaims ++ List("namespace Bottom") ++ bottom ++
      List("end Bottom", "end Auth")
    java.nio.file.Files.writeString(java.nio.file.Path.of(
      "/tmp/claude-0/-home-user-fr-flow/80a77cee-12ee-5106-9f28-3411c00b75ee/scratchpad/r43-spike/rooms-lean-r15/AuthDump.lean"),
      out.mkString("\n\n") + "\n")
  }
